from appium import webdriver
from appium.options.common import AppiumOptions

opts = AppiumOptions()
opts.load_capabilities({
    "platformName": "Windows",
    "appium:automationName": "Wincore",
    "appium:app": "Microsoft.WindowsCalculator_8wekyb3d8bbwe!App",
})
driver = webdriver.Remote("http://127.0.0.1:4723", options=opts)

for key in ("num5Button", "plusButton", "num3Button", "equalButton"):
    driver.find_element("accessibility id", key).click()

# XPath runs natively against the live UIA tree
driver.find_element("xpath", "//*[@AutomationId='CalculatorResults']").text
